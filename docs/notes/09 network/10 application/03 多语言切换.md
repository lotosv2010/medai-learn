# 思路
+ 多语言的实现其实很简单，就是先拿到请求头，然后判断有没有accept-language这个内容，没有的话直接返回默认语言，存在的话进行参数解析，因为存在一个语言的优先级，在我们解析以后做一个从大到小的排序，取第一个语言类型作为返回的参数，然后在返回的时候内容设置改语言。

# 源码
```javascript
const http = require('http')
let languages = {
  en: {
      title: 'hello'
  },
  zh: {
      title: '欢迎'
  }
}

const str2Array = (str) => {
  // 拿到请求头，做分析
  return str.split(',').map(item => {
    const lan = item.split(';');
    return {
      name: lan[0],
      id: lan[1] && +lan[1].split('=')[1] || 1
    }
  }).sort((a, b) =>b.id - a.id); // 为了处理权重，这里倒叙，拿到权重最大的值
}

const server = http.createServer((req, res) => {
  const langs = req.headers['accept-language'];
  // zh-CN,zh;q=0.9,en;q=0.8
  if(!langs) return res.end('Not Found'); // 判断是否支持多语言
  const lanArray = str2Array(langs);
  let r = null;
  // 在本地语言包拿内容
  for (const lan of lanArray) {
    // 如果语言包里有内容，就返回内容
    if(languages[lan.name]) {
      r = languages[lan.name];
      break;
    }
  }
  if(!r) {
    r = languages['en'];
  }
  // 设置请求头 返回需要的语言包
  res.setHeader('Content-Type', 'application/json;charset=utf-8');
  res.end(JSON.stringify(r));
})
server.listen(3000, () => {
  console.log(`server start port 3000`);
});
```
