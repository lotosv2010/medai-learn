# 概述
+ 通过host实现多个网站共用一个端口，多个网站共用一个服务器

# 源码实现
## src
### server.js
```javascript
const http = require('http');
const httpProxy = require('http-proxy');
const proxy = httpProxy.createProxyServer();

const config = {
  'a.gwb.com': 'http://localhost:8089',
  'b.gwb.com': 'http://localhost:8090',
}

const app = http.createServer((req, res) => {
  let host = req.headers['host'];
  host = host?.split(':')[0];
  const target = config[host];
  if(!target) return res.end('NOT FOUND');
  proxy.web(req, res, {
    target
  })
});
app.listen(8088, () => {
  console.log(`server start port 8088`);
})
```

### proxy1.js
```javascript
const http = require('http');
http.createServer((req, res) => {
  res.end('8089');
}).listen(8089);
```

### proxy2.js
```javascript
const http = require('http');
http.createServer((req, res) => {
  res.end('8090');
}).listen(8090);
```

# 测试
## 启动服务器
```shell
cd src
nodemon server.js
```

## 启动代理服务1
```shell
cd src
nodemon proxy1.js
```

## 启动代理服务2
```shell
cd src
nodemon proxy2.js
```
