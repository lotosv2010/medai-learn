# 一、概述
+ 迭代器模式(Iterator Pattern)用于顺序地访问聚合对象内部的元素，又无需知道内部结构，使用了迭代器之后，使用者不需要关心对象的内部结构，就可以按需访问每个元素
+ 提供一种方法顺序一个聚合对象中各个元素，而又不暴露该对象的内部表示。

# 三、代码
```typescript
class Iterator {
  private list: Array<any>;
  private index: number;
  constructor(public container: Container) {
    this.list = container.list;
    this.index = 0;
  }
  next() {
    if (this.hasNext()) {
      return this.list[this.index++];
    }
    return null;
  }
  hasNext() {
    if (this.index >= this.list.length) {
      return false;
    }
    return true;
  }
}

class Container {
  constructor(public list: Array<any>) {}
  getIterator() {
    return new Iterator(this);
  }
}

// 测试代码
let container = new Container([1, 2, 3, 4, 5]);
let iterator = container.getIterator();
while (iterator.hasNext()) {
  console.log(iterator.next());
}
```

# 四、场景
## createIterator
```typescript
function createIterator(array: any[]) {
  let index = 0;
  return {
    next() {
      return index < array.length ? {value: array[index++], done: false} : {value: undefined, done: true}
    }
  }
}
const array: any[] = [1, 2, 3];
const it: any = createIterator(array);
console.log(it.next()); // { value: 1, done: false }
console.log(it.next()); // { value: 2, done: false }
console.log(it.next()); // { value: 3, done: false }
console.log(it.next()); // { value: undefined, done: true }
```

## ES6 <font style="color:rgb(13, 20, 30);">Iterator</font>
+ 在 ES6 中有序集合数据类型有 Array、Map、Set、String、TypedArray、arguments、NodeList
+ 我们需要有一个统一的遍历接口来遍历所有的数据类型
+ 他们都有 `[Symbol.Iterator]` 属性，属性是一个函数，执行函数会返回迭代器
+ 迭代器就有 next 方法顺序返回子元素

```javascript
(Array.prototype as any)[Symbol.iterator] = function() {
  let index = 0;
  return {
    next:() => {
      return index < this.length ?
        {value: this[index++], done: false} :
        {value: undefined, done: true}
    }
  }
}

const arr = [1, 2, 3];
const it = arr[Symbol.iterator]();
console.log(it.next());
console.log(it.next());
console.log(it.next());
console.log(it.next());
```

## 二叉树遍历
+ 二叉树是每个节点最多有两个子树的树结构，通常子树被称作左子树和右子树
    - 根据根节点的顺序可以把遍历分为三种
        * 先序遍历：根节点-->左子树-->右子树
        * 中序遍历：左子树-->根节点-->右子树
        * 后序遍历：左子树-->右子树-->根节点

### 构建树
```typescript
// 树结构类
class Tree {
  constructor(public left: any, public value: any, public right: any) {}
}

// 构建树
function make(array: any[]): Tree {
  if(array.length === 1) return new Tree(null, array[0], null)
  return new Tree(make(array[0]), array[1], make(array[2]));
}

const tree = make([['D', 'B', 'E'], 'A',['F', 'C', 'G']]);

// console.log(tree);
```

### 先序遍历
```typescript
// !先序遍历
// ['A', 'B', 'C', 'D', 'E', 'F', 'G']
function* leftOrder(tree: Tree): any {
  if(tree) {
    yield tree.value;
    yield* leftOrder(tree.left);
    yield* leftOrder(tree.right);
  }
}
// 先序遍历测试
const leftResult: any[] = [];
for (const node of leftOrder(tree)) {
  leftResult.push(node);
}
console.log(leftResult);
```

### 中序遍历
```typescript
// !中序遍历
// ['A', 'B', 'C', 'D', 'E', 'F', 'G']
function* inOrder(tree: Tree): any {
  if(tree) {
    yield* inOrder(tree.left);
    yield tree.value;
    yield* inOrder(tree.right);
  }
}
// 中序遍历测试
const inResult: any[] = [];
for (const node of inOrder(tree)) {
  inResult.push(node);
}
console.log(inResult);

```

### 后序遍历
```typescript
// !后序遍历
// ['A', 'B', 'C', 'D', 'E', 'F', 'G']
function* rightOrder(tree: Tree): any {
  if(tree) {
    yield* rightOrder(tree.left);
    yield* rightOrder(tree.right);
    yield tree.value;
  }
}
// 后序遍历测试
const rightResult: any[] = [];
for (const node of rightOrder(tree)) {
  rightResult.push(node);
}
console.log(rightResult);
```
