# 一、概述
+ 适配器模式又称包装器模式，将一个类的接口转化为用户需要的另一个接口，解决类(对象)之间接口不兼容的问题
+ 旧的接口和使用者不兼容
+ 中间加一个适配器转换接口

# 三、代码
```typescript
// 需要被适配的类
class Socket {
  output() {
    return '220V';
  }
}

abstract class Power {
  abstract charge(): string;
}

class PowerAdapter extends Power {
  constructor(public socket: Socket) {
    super();
  }
  charge(): string {
    return this.socket.output() + ' 转换为 24V';
  } 
}

const p = new PowerAdapter(new Socket);
console.log(p.charge());
```

```typescript
function xhr(config: Record<string, any>): Promise<any> {
  return new Promise((resolve, reject) => {
    const request = new XMLHttpRequest();
    request.open(config.method, config.url, true);
    request.onreadystatechange = function () {
      if(request.readyState == 4) {
        if(request.status == 200) {
          resolve(request.responseText);
        } else {
          reject('Error');
        }
      }
    }
    request.send();
  });
}
function http(config: Record<string, any>): Promise<any> {
  const http = require('http');
  const urlObj = new URL(config.url);
  // console.log(urlObj)
  return new Promise((resolve, reject) => {
    const options = {
      hostname: urlObj.hostname,
      port: urlObj.port,
      path: urlObj.pathname,
      method: config.method
    }
    const req = http.request(options, (response: any) => {
      const chunks: any = [];
      response.on('data', (chunk: any) => {
        chunks.push(chunk);
      });
      response.on('end', () => {
        const result = Buffer.concat(chunks).toString();
        resolve(result);
      });
    });
    req.on('error', (error: any) => {
      console.log(error);
      reject(error);
    });
    req.end();
  })
}
function getDefaultAdaptor(config: Record<string, any>): Promise<any> {
  let adaptor!: Promise<any>;
  if(typeof XMLHttpRequest != 'undefined') {
    adaptor = xhr(config);
  } else if(typeof process != 'undefined') {
    adaptor = http(config);
  }
  return adaptor;
}
function axios(config: Record<string, any>): Promise<any> {
  const adaptor = getDefaultAdaptor(config);
  // 不管是浏览器端还是node端，都是传入一个config对象，返回一个promise
  return adaptor;
}
```

```typescript
function toAxiosAdaptor(options) {
  return axios({
    url: options.url,
    method: options.type
  }).then(options.success, options.error);

$.ajax = function(options) {
```

```typescript
function promisify(fn: any) {
  return function (filename: string, encoding: string) {
    return new Promise((resolve, reject) => {
      fn(filename, encoding, (error: Error, data: any) => {
        if(error) {
          reject(error);
        } else {
          resolve(data);
        }
      })
    });
  }
}
```
