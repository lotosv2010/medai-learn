# 一、概述
+ 执行命令时，发布者和执行者分开
+ 中间加入命令对象，作为中转站
+ 三种角色
    - Receiver 接受者角色：该角色就是干活的角色，命令传递到这里是应该被执行的
    - Command 命令角色：需要执行的所有命令都在这里声明
    - Invoker 调用者角色：接收到命令，并执行命令

# 三、代码
```typescript
class Cooker {
  cook() {
    console.log('cook');
  }
}

class Cleaner{
  clean() {
    console.log('clean');
  }
}

abstract class Command {
  abstract execute(): void;
}

class CookCommand extends Command{
  private cooker: Cooker;
  constructor(receiver: Cooker) {
    super();
    this.cooker = receiver;
  }
  execute() {
    this.cooker.cook();
  }
}

class CleanCommand extends Command{
  private cleaner: Cleaner;
  constructor(receiver: Cleaner) {
    super();
    this.cleaner = receiver;
  }
  execute() {
    this.cleaner.clean();
  }
}

class Customer{
  private command: Command;
  constructor(command: Command) {
    this.command = command;
  }
  setCommand(command: Command) {
    this.command = command;
  }
  cook() {
    this.command.execute();
  }
  clean() {
    this.command.execute();
  }
}

const cooker = new Cooker();
const cleaner = new Cleaner();
const cookCommand = new CookCommand(cooker);
const cleanCommand = new CleanCommand(cleaner);
const c = new Customer(cookCommand);
c.cook();
c.setCommand(cleanCommand);
c.clean();
```

## 多步撤销和重做
```html
<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta http-equiv="X-UA-Compatible" content="IE=edge">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>命令模式</title>
</head>
<body>
  <p id="number">0</p>
  <button id="addBtn">+</button>
  <button id="undo">undo</button>
  <button id="redo">redo</button>
  <script>
    const worker = {
      history: [],
      index: -1, // 指针
      add() {
        const oldValue = parseInt(number.innerHTML);
        const newValue = oldValue + 1;
        this.history.push(newValue);
        this.index = this.history.length - 1;
        number.innerHTML = newValue;
      },
      undo() {
        number.innerHTML = this.index > 0 ? this.history[--this.index] : 0;
      },
      redo() {
        const len = this.history.length - 1;
        number.innerHTML = this.index < len ? this.history[this.index++] : this.history[len];
      }
    }
    class AddCommand{
      constructor(receiver) {
        this.receiver = receiver;
      }
      execute() {
        this.receiver.add();
      }
    }
    class UndoCommand{
      constructor(receiver) {
        this.receiver = receiver;
      }
      execute() {
        this.receiver.undo();
      }
    }
    class RedoCommand{
      constructor(receiver) {
        this.receiver = receiver;
      }
      execute() {
        this.receiver.redo();
      }
    }
    const addCommand = new AddCommand(worker);
    const undoCommand = new UndoCommand(worker);
    const redoCommand = new RedoCommand(worker);
    addBtn.addEventListener('click', () => addCommand.execute());
    undo.addEventListener('click', () => undoCommand.execute());
    redo.addEventListener('click', () => redoCommand.execute());
  </script>
</body>
</html>
```
