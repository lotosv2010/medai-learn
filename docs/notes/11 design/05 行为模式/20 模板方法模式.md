# 一、概述
+ 模板方法模式在一个方法中定义一个算法骨架，而将一些步骤的实现延迟到子类中
+ 模板方法使得子类可以在不改变算法结构的情况下，重新定义算法中某些步骤的具体实现
+ 一般有两部分组成，第一部分是抽象父类，第二部分是具体的实现子类
+ 好莱坞原则，子类放弃了控制权，改由父类调用
    - 发布订阅
    - 回调函数

# 三、代码
```typescript
abstract class Person {
  dinner() {
    // 这个顺序是固定的，不能随意改变
    this.buy();  // 买菜
    this.cook(); // 做饭
    this.eat();  // 吃饭
  }
  abstract buy(): void;
  abstract cook(): void;
  abstract eat(): void;
}

class Man extends Person{
  buy(): void {
    console.log('买黄瓜');
  }
  cook(): void {
    console.log('拍黄瓜');
  }
  eat(): void {
    console.log('吃拍黄瓜');
  }
}
class Women extends Person{
  buy(): void {
    console.log('买西红柿');
  }
  cook(): void {
    console.log('糖拌西红柿');
  }
  eat(): void {
    console.log('吃糖拌西红柿');
  }
}

const man = new Man();
man.dinner();
const women = new Women();
women.dinner();
```

    class Modal{
      constructor(options) {
        this.title = options.title || '标题';
        this.content = options.content || '内容';
        this.confirmBtnText = options.confirmBtnText || '确认';
        this.cancelBtnText = options.cancelBtnText || '取消';
        this.onConfirm = options.onConfirm;
        this.onCancel = options.onCancel;
        this.init(); // 初始化
        this.eventListener(); // 绑定事件
      }
      // 初始化，用于在界面上绘制一个对话框
      init() {
        this.panel = document.createElement('div');
        this.panel.className = 'modal';
        // 标题
        const titleP = document.createElement('p');
        titleP.innerHTML = this.title;
        this.panel.appendChild(titleP);
        // 内容
        const contentP = document.createElement('p');
        contentP.innerHTML = this.content;
        this.panel.appendChild(contentP);
        // footer
        const footer = this.footer = document.createElement('footer');
        footer.className = 'footer';
        this.panel.appendChild(footer);
        // 确认按钮
        const confirmBtn = this.confirmBtn = document.createElement('button');
        confirmBtn.className = 'button confirm-btn';
        confirmBtn.innerHTML = this.confirmBtnText;
        footer.appendChild(confirmBtn);
        // 取消按钮
        const cancelBtn = this.cancelBtn = document.createElement('button');
        cancelBtn.className = 'button cancel-btn';
        cancelBtn.innerHTML = this.cancelBtnText;
        footer.appendChild(cancelBtn);
        // document.body.appendChild(this.panel);
      }
      eventListener() {
        this.confirmBtn.addEventListener('click', (event) => {
          this.onConfirm && this.onConfirm({isClose: true});
          this.hide();
        });
        this.cancelBtn.addEventListener('click', (event) => {
          this.onCancel && this.onCancel({isClose: true});
          this.hide();
        });
      }
      hide() {
        // this.panel.style.display = 'none';
        this.panel.parentNode.removeChild(this.panel);
      }
      show() {
        // this.panel.style.display = 'block';
        document.body.append(this.panel);
      }
    }
    class ConfirmModal extends Modal{}
    class InfoModal extends Modal {
      constructor(options) {
        super(options);
        this.footer.removeChild(this.cancelBtn);
        this.panel.removeChild(this.panel.firstChild);
      }
    }
    const confirmDialog = new ConfirmModal({
      title: '温馨提示',
      content: '确认要删除吗？',
      onConfirm: (data) => {
        console.log(data);
      }
    });
    const infoDialog = new InfoModal({
      content: '这是一条基本信息',
      confirmBtnText: '知道了',
      onConfirm: (data) => {
        console.log(data);
      }
    });
    function showConfirm() {
      confirmDialog.show();
    }
    function showInfo() {
      infoDialog.show();
    }
  </script>
