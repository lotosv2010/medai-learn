# 一、概述
+ 将抽象部分与它的实现部分分离，这样抽象化与实现化解耦，使它们可以独立的变化
+ 应用场景是实现系统可能有多个角度分类，每一种角度都可能变化
+ 桥方可以通过实现桥接口进行单方面扩展，而另一方可以继续抽象类而单方面扩展，而之间的调用就从桥接口来作为突破口，不会受到双方扩展的任何影响

# 三、代码
```typescript
abstract class A {
  constructor(private bridge: B) {}
  go() {
    console.log(`从${this.from()}到达${this.bridge.to()}`);
  }
  abstract from(): string;
}

class A1 extends A {
  from(): string {
    return 'A1';
  }
}
class A2 extends A {
  from(): string {
    return 'A2';
  }
}
class A3 extends A {
  from(): string {
    return 'A3';
  }
}

abstract class B {
  abstract to(): string;
}

class B1 extends B {
  to(): string {
    return 'B1';
  }
}
class B2 extends B {
  to(): string {
    return 'B2';
  }
}
class B3 extends B {
  to(): string {
    return 'B3';
  }
}

const b2 = new B2();
const a1 = new A1(b2);
a1.go();
const a2 = new A2(b2);
a2.go();
```

    name: `bridge${id}`
  });

app.use('/', (req, res) => {

<!DOCTYPE html>
<html lang="en">

  <meta name="viewport" content="width=device-width, initial-scale=1.0">

      text-align: center;
    }
    canvas {
      border: 1px solid #ccc;
    }
  </style>

<body>
  <canvas id="canvas" width="1000px" height="600px"></canvas>
  <script>
    const canvas = document.getElementById('canvas');
    const ctx = canvas.getContext('2d');
    function Position(x, y) {
      this.x = x;
      this.y = y;
    }
    function Color(color) {
      this.color = color;
    }
    function Circle(x, y, color, radius) {
      this.position = new Position(x, y);
      this.color = new Color(color);
      this.radius = radius;
    } 
    Circle.prototype.render = function() {
      ctx.beginPath();
      ctx.arc(this.position.x, this.position.y, this.radius, 0, 2 * Math.PI);
      ctx.fillStyle = this.color.color; // getRandomColor();
      ctx.fill();
    }
    const c = new Circle(200, 200, '#25b864', 100);
    c.render();
  </script>
</body>
</html>
