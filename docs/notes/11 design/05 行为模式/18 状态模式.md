# 一、概述
+ 当一个对象的内部状态发生改变时，会导致其行为的改变，这看起来像是改变了对象
+ 对象有自己的状态
+ 不同的状态下执行的逻辑不一样
+ 明确状态和每个状态下执行的动作
+ 用来减少 `if...else...`	子句

abstract class Status {
  abstract show(): void;

class Battery {
  private amount: number = 100;
  public status!: Status;
  constructor() {
    this.status = new SuccessState(this);
  }
  show() {
    
    if(this.amount >= 80) {
      // this.amount = 90;
      this.status = new SuccessState(this);
    } else if(this.amount >= 20 && this.amount < 80) {
      // this.amount = 50;
      this.status = new WaringState(this);
    } else if(this.amount <= 20) {
      // this.amount = 10;
      this.status = new ErrorState(this);

    this.status.show();

    this.amount = amount;

}

      }
    }

      render(element) {
        element.innerHTML = '取消'
      }
    }
    class Button {
      constructor(container) {
        this.liked = false; // 按钮的默认是未点赞状态
        this.state = likeState; // 设置当前的状态为点赞
        this.element = document.createElement('button');
        container.appendChild(this.element);
        this.render(); // 渲染按钮本身
      }
      render(){
        this.state.render(this.element);
      }
      setState(state) {
        this.state = state; // 修改状态
        this.render(); // 修改完状态重新渲染
      }
    }
    const button = new Button(document.body);
    button.element.addEventListener('click', () => {
      button.liked = !button.liked;
      button.setState(button.liked ? likedState : likeState);

## promise
```typescript
enum State{
  PENDING = 'PENDING',     // 等待状态
  FULFILLED = 'FULFILLED', // 成功状态
  REJECTED = 'REJECTED'    // 失败状态
}

class Promise {
  private state = State.PENDING;
  private success: Array<Function> = [];
  private error: Array<Function> = []
  private value: any = undefined
  private reason: any = undefined
  constructor(executor: (resolve: Function, reject: Function) => any) {
    const resolve = (data: any) => {
      if(this.state === State.PENDING) {
        this.value = data;
        this.state = State.FULFILLED;
        this.success.forEach(fn => fn()); // 发布的过程
      }
    }
    const reject = (error: any) => {
      if(this.state === State.PENDING) {
        this.reason = error;
        this.state = State.REJECTED;
        this.error.forEach(fn => fn());
      }
    }
    try {
      executor(resolve, reject);
    } catch (error) {
      reject(error);
    }
  }
  then(onFulFilled: Function, onReject: Function) {
    if(this.state === State.FULFILLED) {
      onFulFilled(this.value);
    }
    if(this.state === State.REJECTED) {
      onReject(this.reason);
    }
    if(this.state == State.PENDING) { // 订阅的过程
      this.success.push(() => {
        // todo
        onFulFilled(this.value)
      })
      this.error.push(() => {
        onReject(this.reason)
      })
    }
  }
}

const promise = new Promise((resolve, reject) => {
  const random =  Math.random();
  random > 0.5 ? resolve(`data: ${random}`) : reject(`error: ${random}`);
}).then((data: any) => {
  console.log(data);
}, (error: any) => {
  console.log(error)
});
```

## 有限状态机
+ 事物拥有多种状态，任意时间只会处于一种状态不会处于多种状态
+ 动作可以改变事物状态，一个动作可以通过条件判断，改变事物到不同的状态，但是不能同时指向多个状态
+ 状态总数是有限的
+ javascript-state-machine
    - from：当前行为从哪个状态来
    - to：当前行为执行完会过渡到那个状态
    - name：当前行为的名字
+ fsm.can(t) - return true：如果过渡方法 t 可以从当前状态触发
+ fsm.cannot(t) - return true：如果当前状态下不能发生过渡方法 t
+ fsm.transitions()：返回当前状态可以过渡到的状态列表
+ fsm.allTransitions()：返回所有过渡方法的列表
+ fsm.allStates()：返回状态机有的所有状态列表
+ onBefore：在特定动作TRANSITION之前触发
+ onLeaveState：离开任何一个状态时触发
+ onLeave：在离开特定状态STATE时触发
+ onTransition：在任何动作发生期间触发
+ onEnterState：当进入任何状态时触发
+ onEnter：进入一个特定的状态STATE时触发
+ on：onEnter的简写
+ onAfterTransition：在任何动作触发之后触发
+ onAfter：在特定动作TRANSITION后触发
+ on：onAfter的简写

```javascript
// const StateMachine = require('javascript-state-machine');

class StateMachine {
  constructor(options) {
    // init定义初始状态
    // transitions定义状态转化规则
    // methods定义监听函数
    const {init='', transitions=[], methods={}} = options;
    this.state = init;
    transitions.forEach(transition => {
      const {from, to, name} = transition;
      const onMethod = `on${name.slice(0,1).toUpperCase()}${name.slice(1)}`;
      this[onMethod] = function() {
        this.state = to;
        methods[onMethod] && methods[onMethod]();
      }
    });
  }
}

const fsm = new StateMachine({
  init: 'solid',
  transitions: [
    { name: 'melt',     from: 'solid',  to: 'liquid' },
    { name: 'freeze',   from: 'liquid', to: 'solid'  },
    { name: 'vaporize', from: 'liquid', to: 'gas'    },
    { name: 'condense', from: 'gas',    to: 'liquid' }
  ],
  methods: {
    onMelt:     function() { console.log('I melted')    }, // 监听融化的动作，当融化时触发
    onFreeze:   function() { console.log('I froze')     },
    onVaporize: function() { console.log('I vaporized') },
    onCondense: function() { console.log('I condensed') }
  }
});
// console.log(fsm.allStates());
fsm.onMelt();
fsm.onFreeze();
fsm.onVaporize();
fsm.onCondense();
```
