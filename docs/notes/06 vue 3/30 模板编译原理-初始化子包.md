# 创建子包
```shell
cd g-vue-next
mkdir compiler-core compiler-dom
```

## compiler-core
### 创建入口
```shell
cd packages/compiler-dom
mkdir src
cd src
touch index.ts
```

pnpm init

# 源码入口
## packages
### compiler-dom
#### src
##### index.ts
```typescript
export * from '@g-vue-next/compiler-core'
```

### compiler-core
#### src
##### index.ts
```tsx
export { baseParse } from './parser'
export { transform } from './transform'
export { generate } from './codegen'
export { baseCompile } from './compile'
```

##### parser.ts
```typescript
export function baseParse(input: string, options?: any) {}
```

##### transform.ts
```typescript
export function transform(root: any, options: any) {}
```

##### codegen.ts
```typescript
export function generate(
  ast: any,
  options: any = {},
): any {}
```

##### compile.ts
```typescript
import { CompilerOptions } from "./options";

export function baseCompile(
  source: string | any,
  options: CompilerOptions = {},
) {}
```

##### options.ts
```typescript
export type CompilerOptions = any
```
